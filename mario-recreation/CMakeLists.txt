cmake_minimum_required(VERSION 3.16)
project(mario_recreation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(APPLE)
  execute_process(COMMAND brew --prefix OUTPUT_VARIABLE BREW_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(BREW_PREFIX)
    set(ENV{PKG_CONFIG_PATH} "${BREW_PREFIX}/opt/readline/lib/pkgconfig:${BREW_PREFIX}/lib/pkgconfig:$ENV{PKG_CONFIG_PATH}")
  endif()
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2 SDL2_image SDL2_ttf SDL2_mixer)
file(GLOB_RECURSE ENGINE CONFIGURE_DEPENDS vendor/src/*.cpp)
list(FILTER ENGINE EXCLUDE REGEX "/main.cpp$")
add_executable(mario native/main.cpp ${ENGINE})
target_include_directories(mario PRIVATE vendor/include native)
target_link_libraries(mario PRIVATE PkgConfig::SDL)
target_compile_definitions(mario PRIVATE MARIO_RESOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/vendor")
option(MARIO_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
if(MARIO_SANITIZE)
  target_compile_options(mario PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mario PRIVATE -fsanitize=address,undefined)
endif()
